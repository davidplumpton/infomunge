// MatcherResult represents the result of an assertion
type MatcherResult struct {
        Success bool
        Message string
}

// Matcher represents a matcher function that validates a value
type Matcher func(interface{}) *MatcherResult

// MATCHED is a constant that represents a successful match
var MATCHED = &MatcherResult{Success: true, Message: ""}

// assertion functions return Matcher objects that can be applied to values

// typeError creates a failed MatcherResult for type mismatch
func typeError(expected string, value interface{}) *MatcherResult {
        return &MatcherResult{Success: false, Message: fmt.Sprintf("expected %s, got %T", expected, value)}
}

// typeMatcherFactory creates a type matcher that validates via type assertion
func typeMatcherFactory(typeName string, checker func(interface{}) bool) Matcher {
        return func(value interface{}) *MatcherResult {
                if checker(value) {
                        return MATCHED
                }
                return typeError(typeName, value)
        }
}

func beEmpty() Matcher {
        return func(value interface{}) *MatcherResult {
                switch v := value.(type) {
                case string:
                        if len(v) == 0 {
                                return MATCHED
                        }
                        return &MatcherResult{Success: false, Message: fmt.Sprintf("expected empty string, got %q", v)}
                case []interface{}:
                        if len(v) == 0 {
                                return MATCHED

                        }
                        return &MatcherResult{Success: false, Message: fmt.Sprintf("expected empty object, got object with %d keys", len(v))}
                default:
                        return &MatcherResult{Success: false, Message: fmt.Sprintf("expected string, array, or object, got %T", value)}
                }
        }

// Helper functions

func isEqual(a, b interface{}) bool {
        // Handle nil cases
        if a == nil && b == nil {
                return true
        }
        if a == nil || b == nil {
                return false
        }

        // For numbers, compare as float64
        aNum, aIsNum := toNumberForAssertion(a)
        bNum, bIsNum := toNumberForAssertion(b)
        if aIsNum && bIsNum {
                return aNum == bNum
        }

        // For strings
        aStr, aIsStr := a.(string)
        bStr, bIsStr := b.(string)
        if aIsStr && bIsStr {
                return aStr == bStr
        }

        // For booleans
        aBool, aIsBool := a.(bool)
        bBool, bIsBool := b.(bool)
        if aIsBool && bIsBool {
                return aBool == bBool
        }

        // For arrays
        aArr, aIsArr := a.([]interface{})
        bArr, bIsArr := b.([]interface{})
        if aIsArr && bIsArr {
                if len(aArr) != len(bArr) {
                        return false
                }
                for i := range aArr {
                        if !isEqual(aArr[i], bArr[i]) {
                                return false
                        }
                }
                return true
        }

        // For objects
        aObj, aIsObj := a.(map[string]interface{})
        bObj, bIsObj := b.(map[string]interface{})
        if aIsObj && bIsObj {
                if len(aObj) != len(bObj) {
                        return false
                }
                for key, val := range aObj {
                        bVal, exists := bObj[key]
                        if !exists || !isEqual(val, bVal) {
                                return false
                        }
                }
                return true
        }

        return false
}

func toNumberForAssertion(v interface{}) (float64, bool) {
        switch n := v.(type) {
        case float64:
                return n, true
        case int:
                return float64(n), true
        default:
                return 0, false
        }
}

func compareValues(a, b interface{}) (int, error) {
        aNum, aIsNum := toNumberForAssertion(a)
        bNum, bIsNum := toNumberForAssertion(b)
        if !aIsNum || !bIsNum {
                return 0, fmt.Errorf("cannot compare non-numeric values: %T and %T", a, b)
        }

        if aNum < bNum {
                return -1, nil
        }
        if aNum > bNum {
                return 1, nil
        }
        return 0, nil
}

        cond, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err

        condBool, ok := cond.(bool)
        if !ok {

// DefaultLoopTimeout is the default maximum duration for a while loop if no deadline is set.
const DefaultLoopTimeout = 30 * time.Second

// callBuiltinWhile implements the __while(condition, body) function.
// Executes the body repeatedly while the condition is true.
// Both condition and body are passed as unevaluated AST expressions.
// Supports break and continue statements via ControlFlowSignal.
// Checks for timeout via __deadline in context or uses DefaultLoopTimeout.
// getDeadline retrieves the loop deadline from context or uses default timeout
func getDeadline(context map[string]interface{}, startTime time.Time) time.Time {
        if d, ok := context["__deadline"].(time.Time); ok {
                return d
        }
        return startTime.Add(DefaultLoopTimeout)
}

// checkTimeout verifies the loop hasn't exceeded the deadline
func checkTimeout(i int, startTime, deadline time.Time, e *ast.CallExpr) error {
        // Check for timeout every 100 iterations to balance overhead and responsiveness
        if i%100 == 0 && time.Now().After(deadline) {
                return newPosError(fmt.Sprintf("while loop timed out after %v", time.Since(startTime)), e.Pos())
        }
        return nil
}

// evaluateCondition evaluates the while condition and converts it to boolean
func evaluateCondition(e *ast.CallExpr, context map[string]interface{}, depth int) (bool, error) {
        cond, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return false, err
        }

        condBool, ok := cond.(bool)
        if !ok {
                return false, newPosError(fmt.Sprintf("while condition must be a boolean, got %T", cond), e.Args[0].Pos())
        }
        return condBool, nil
}

// evaluateBody executes the while loop body and returns the result or control flow signal
func evaluateBody(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        return evalASTWithDepth(e.Args[1], context, depth)
}

// processBodyResult handles control flow signals from the loop body
func processBodyResult(bodyResult interface{}) (*ControlFlowSignal, bool) {
        if signal, ok := bodyResult.(*ControlFlowSignal); ok {
                return signal, true
        }
        return nil, false
}

func callBuiltinWhile(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("while loop requires exactly 2 arguments: condition, body", e.Pos())
        }

        var result interface{}
        startTime := time.Now()
        deadline := getDeadline(context, startTime)

loop:
        for i := 0; ; i++ {
                // Check for timeout
                if err := checkTimeout(i, startTime, deadline, e); err != nil {
                        return nil, err
                }

                // Evaluate condition
                condBool, err := evaluateCondition(e, context, depth)
                if err != nil {
                        return nil, err
                }

                // Exit loop if condition is false
                if !condBool {
                        break
                }

                // Execute body
                bodyResult, err := evaluateBody(e, context, depth)
                if err != nil {
                        return nil, err
                }

                // Check for control flow signals
                signal, isSignal := processBodyResult(bodyResult)
                if isSignal {
                        switch signal.Type {
                        case "break":
                                break loop
                        case "continue":
                                continue loop
                        }
                }

                // Only update result if it's not a signal
                if !isSignal {
                        result = bodyResult
                }
        }

        return result, nil
}

// callBuiltinAssign implements the __assign(varName, value) function.
// This modifies the context map to update the variable and returns the assigned value.
// Used to enable assignment expressions in loops like: x = x + 1
func callBuiltinAssign(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("assignment requires exactly 2 arguments: variable name and value", e.Pos())
        }

        // First argument must be a string literal (variable name)
        varNameExpr, ok := e.Args[0].(*ast.BasicLit)
        if !ok || varNameExpr.Kind != token.STRING {
                return nil, newPosError("assignment variable name must be a string literal", e.Args[0].Pos())
        }

        varName, err := strconv.Unquote(varNameExpr.Value)
        if err != nil {
                return nil, newPosError(fmt.Sprintf("invalid variable name: %s", err), e.Args[0].Pos())
        }

        // Evaluate the value expression
        value, err := evalASTWithDepth(e.Args[1], context, depth)
        if err != nil {
                return nil, err
        }

        // Update the context with the new value
        context[varName] = value

        // Return the assigned value
        return value, nil
}

// callBuiltinSeq implements the __seq(expr1, expr2, ...) function.
// Evaluates all expressions in order and returns the last one.
// Used to handle multiple statements in sequence, e.g.: while(...); finalExpr
// If any expression returns a control flow signal (break/continue), propagates it immediately.
func callBuiltinSeq(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) < 1 {
                return nil, newPosError("seq requires at least 1 argument", e.Pos())
        }

        var result interface{}
        var err error

        // Evaluate all arguments in order
        for _, arg := range e.Args {
                result, err = evalASTWithDepth(arg, context, depth)
                if err != nil {
                        return nil, err
                }
                
                // If a control flow signal is encountered, propagate it immediately
                if _, ok := result.(*ControlFlowSignal); ok {
                        return result, nil
                }
        }

        // Return the last evaluated result
        return result, nil
}

        left, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err

        return evalASTWithDepth(e.Args[1], context, depth)
}

// callBuiltinLambdaAST implements the __lambda(paramNames, body) function.
// paramNames is a string literal of comma-separated parameter names.
// body is an unevaluated expression that will be stored and evaluated later.
func callBuiltinLambdaAST(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("lambda expression requires exactly 2 arguments: paramNames, body", e.Pos())
        }

        // Evaluate the first argument (parameter names) - should be a string literal
        paramNamesVal, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        paramNamesStr, ok := paramNamesVal.(string)
        if !ok {
                return nil, newPosError("lambda parameter names must be a string", e.Pos())
        }

        // Parse parameter names
        var paramNames []string
        if paramNamesStr != "" {
                for _, p := range strings.Split(paramNamesStr, ",") {
                        paramNames = append(paramNames, strings.TrimSpace(p))
                }
        }

        // Store the body as an unevaluated AST expression
        bodyExpr := e.Args[1]
        bodyStr := exprToString(bodyExpr)

        return &Lambda{
                ParamNames: paramNames,
                Body:       bodyStr,
                BodyAST:    bodyExpr, // Store the AST for later evaluation
        }, nil
}

// callBuiltinFilter implements the __filter(array, lambda) function.
// It filters an array using a lambda function that returns true for elements to keep.
func callBuiltinFilter(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("filter requires exactly 2 arguments: array, lambda", e.Pos())
        }

        // Evaluate the array argument
        arrayVal, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        // Check that the first argument is an array
        array, ok := arrayVal.([]interface{})
        if !ok {
                return nil, newPosError(fmt.Sprintf("filter expects an array, got %T", arrayVal), e.Args[0].Pos())
        }

        // Evaluate the lambda argument to get the lambda function
        lambdaVal, err := evalASTWithDepth(e.Args[1], context, depth)
        if err != nil {
                return nil, err
        }

        lambda, ok := lambdaVal.(*Lambda)
        if !ok {
                return nil, newPosError(fmt.Sprintf("filter expects a lambda function, got %T", lambdaVal), e.Args[1].Pos())
        }

        // The lambda should have 1 parameter (the element) or 2 (element and index)
        if len(lambda.ParamNames) < 1 || len(lambda.ParamNames) > 2 {
                return nil, newPosError(fmt.Sprintf("filter lambda must have 1 or 2 parameters, got %d", len(lambda.ParamNames)), e.Args[1].Pos())
        }

        result := make([]interface{}, 0, len(array))
        for i, elem := range array {
                lambdaContext := copyContext(context)
                lambdaContext[lambda.ParamNames[0]] = elem
                if len(lambda.ParamNames) > 1 {
                        lambdaContext[lambda.ParamNames[1]] = i
                }

                condVal, err := evalASTWithDepth(lambda.BodyAST, lambdaContext, depth+1)
                if err != nil {
                        return nil, err
                }

                // Convert condition to boolean
                condBool, ok := condVal.(bool)
                if !ok {
                        return nil, newPosError(fmt.Sprintf("filter lambda must return a boolean, got %T", condVal), e.Args[1].Pos())
                }

                // If the condition is true, include the element
                if condBool {
                        result = append(result, elem)
                }
        }

        return result, nil
}

// callBuiltinReduce implements the __reduce(array, lambda) function.
// It reduces an array to a single value by applying a lambda function with an accumulator.
// The lambda receives (accumulator, element) or (accumulator, element, index).
// For the first element, the accumulator is the first array element.
func callBuiltinReduce(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("reduce requires exactly 2 arguments: array, lambda", e.Pos())
        }

        // Evaluate the array argument
        arrayVal, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        // Check that the first argument is an array
        array, ok := arrayVal.([]interface{})
        if !ok {
                return nil, newPosError(fmt.Sprintf("reduce expects an array, got %T", arrayVal), e.Args[0].Pos())
        }

        // Check that array is not empty
        if len(array) == 0 {
                return nil, newPosError("reduce cannot be applied to an empty array", e.Args[0].Pos())
        }

        // Evaluate the lambda argument to get the lambda function
        lambdaVal, err := evalASTWithDepth(e.Args[1], context, depth)
        if err != nil {
                return nil, err
        }

        lambda, ok := lambdaVal.(*Lambda)
        if !ok {
                return nil, newPosError(fmt.Sprintf("reduce expects a lambda function, got %T", lambdaVal), e.Args[1].Pos())
        }

        // The lambda should have 2 parameters (accumulator and element) or 3 (accumulator, element, and index)
        if len(lambda.ParamNames) < 2 || len(lambda.ParamNames) > 3 {
                return nil, newPosError(fmt.Sprintf("reduce lambda must have 2 or 3 parameters, got %d", len(lambda.ParamNames)), e.Args[1].Pos())
        }

        accumulator := array[0]

        for i := 1; i < len(array); i++ {
                lambdaContext := copyContext(context)
                lambdaContext[lambda.ParamNames[0]] = accumulator
                lambdaContext[lambda.ParamNames[1]] = array[i]
                if len(lambda.ParamNames) > 2 {
                        lambdaContext[lambda.ParamNames[2]] = i
                }

                result, err := evalASTWithDepth(lambda.BodyAST, lambdaContext, depth+1)
                if err != nil {
                        return nil, err
                }
                accumulator = result
        }

        return accumulator, nil
}

// exprToString converts an AST expression back to a string representation.
func exprToString(expr ast.Expr) string {
        switch e := expr.(type) {
        case *ast.BasicLit:
                return e.Value
        case *ast.Ident:
                return e.Name
        case *ast.BinaryExpr:
                left := exprToString(e.X)
                right := exprToString(e.Y)
                return fmt.Sprintf("%s %s %s", left, e.Op, right)
        case *ast.ParenExpr:
                return fmt.Sprintf("(%s)", exprToString(e.X))
        case *ast.CallExpr:
                fun := exprToString(e.Fun)
                var args []string
                for _, arg := range e.Args {
                        args = append(args, exprToString(arg))
                }
                return fmt.Sprintf("%s(%s)", fun, strings.Join(args, ", "))
        case *ast.IndexExpr:
                x := exprToString(e.X)
                index := exprToString(e.Index)
                return fmt.Sprintf("%s[%s]", x, index)
        case *ast.CompositeLit:
                return fmt.Sprintf("{...}")
        default:
                return fmt.Sprintf("%v", expr)
        }
}

// callBuiltinSort implements the sort(array) function.
// It sorts an array of comparable elements (numbers or strings).
func callBuiltinSort(args []interface{}, e *ast.CallExpr) (interface{}, error) {
        if len(args) != 1 {
                return nil, newPosError("sort requires exactly 1 argument", e.Pos())
        }

        arr, ok := args[0].([]interface{})
        if !ok {
                return nil, newPosError(fmt.Sprintf("sort expects an array, got %T", args[0]), e.Pos())
        }

        // Create a copy to avoid mutating the original
        result := make([]interface{}, len(arr))
        copy(result, arr)

        // Check if all elements are numbers or all are strings
        allNumbers := true
        allStrings := true

        for _, item := range result {
                switch item.(type) {
                case int, float64:
                        allStrings = false
                case string:
                        allNumbers = false
                default:
                        return nil, newPosError(fmt.Sprintf("sort only supports arrays of numbers or strings, got %T", item), e.Pos())
                }
        }

        if allNumbers {
                // Sort numbers
                sortInterface(result, func(a, b interface{}) bool {
                        af, _ := toFloat64(a)
                        bf, _ := toFloat64(b)
                        return af < bf
                })
        } else if allStrings {
                // Sort strings
                sortInterface(result, func(a, b interface{}) bool {
                        return a.(string) < b.(string)
                })
        }

        return result, nil
}

// Helper function for sorting using a comparison function
func sortInterface(arr []interface{}, less func(a, b interface{}) bool) {
        for i := 0; i < len(arr); i++ {
                for j := i + 1; j < len(arr); j++ {
                        if less(arr[j], arr[i]) {
                                arr[i], arr[j] = arr[j], arr[i]
                        }
                }
        }
}

        if !ok {

        for k, v := range left {

        for k, v := range right {
                result[k] = v
        }

        return result, nil
}

// callBuiltinModCall implements __modcall("Module", "func", args...) for module function calls.
func callBuiltinModCall(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) < 2 {
                return nil, newPosError("__modcall requires at least 2 arguments: module name and function name", e.Pos())
        }

        modNameVal, err := evalASTWithDepth(e.Args[0], context, depth+1)
        if err != nil {
                return nil, err
        }
        funcNameVal, err := evalASTWithDepth(e.Args[1], context, depth+1)
        if err != nil {
                return nil, err
        }

        modName, ok1 := modNameVal.(string)
        funcName, ok2 := funcNameVal.(string)
        if !ok1 || !ok2 {
                return nil, newPosError("__modcall requires string module and function names", e.Pos())
        }

        modVal, ok := context[modName]
        if !ok {
                return nil, newPosError(fmt.Sprintf("module %q not found", modName), e.Pos())
        }

        ns, ok := modVal.(map[string]interface{})
        if !ok {
                return nil, newPosError(fmt.Sprintf("%q is not a module namespace", modName), e.Pos())
        }

        fnVal, ok := ns[funcName]
        if !ok {
                return nil, newPosError(fmt.Sprintf("module %q has no function %q", modName, funcName), e.Pos())
        }

        l, ok := fnVal.(*Lambda)
        if !ok {
                return nil, newPosError(fmt.Sprintf("%s::%s is not a function", modName, funcName), e.Pos())
        }

        args := make([]interface{}, 0, len(e.Args)-2)
        for _, argExpr := range e.Args[2:] {
                v, err := evalASTWithDepth(argExpr, context, depth+1)
                if err != nil {
                        return nil, err
                }
                args = append(args, v)
        }

        return callUserLambda(l, args, context, depth+1)
}

// callUserLambda invokes a user-defined Lambda with the given arguments.
func callUserLambda(l *Lambda, args []interface{}, callingContext map[string]interface{}, depth int) (interface{}, error) {
        base := make(map[string]interface{})

        if l.Env != nil {
                for k, v := range l.Env {
                        base[k] = v
                }
        }

        for k, v := range callingContext {
                base[k] = v
        }

        for i, name := range l.ParamNames {
                if i < len(args) {
                        base[name] = args[i]
                }
        }

        return evalASTWithDepth(l.BodyAST, base, depth)
}

// It recursively descends through maps and arrays, collecting all values
// where an object has a key equal to fieldName.
func callBuiltinDeep(args []interface{}, e *ast.CallExpr) (interface{}, error) {
        if len(args) != 2 {
                return nil, newPosError("recursive descent (..) requires exactly 2 arguments: root, fieldName", e.Pos())
        }

        fieldName, err := assertStringArg(args[1], 2, "__deep", e)
        if err != nil {
                return nil, err
        }

        result := make([]interface{}, 0)
        if err := deepCollect(args[0], fieldName, &result, 0, e.Pos()); err != nil {
                return nil, err
        }
        return result, nil
}

// deepCollect recursively collects all values for the given field name.
func deepCollect(node interface{}, field string, out *[]interface{}, depth int, pos token.Pos) error {
        if depth > maxDeepDepth {
                return newPosError("deep search depth limit exceeded", pos)
        }

        switch v := node.(type) {
        case map[string]interface{}:
                // If object has the field, add it to results
                if val, ok := v[field]; ok {
                        *out = append(*out, val)
                }
                // Recurse into all values
                for _, child := range v {
                        if err := deepCollect(child, field, out, depth+1, pos); err != nil {
                                return err
                        }
                }

        case []interface{}:
                // Recurse into each array element
                for _, elem := range v {
                        if err := deepCollect(elem, field, out, depth+1, pos); err != nil {
                                return err
                        }
                }
        }

        return nil
}

// callBuiltinCoerce implements the __coerce(value, TypeExpr) function for the "as" operator.
// It coerces a value to the specified type expression. Supports built-in types (String, Number, Boolean),
// custom types defined with the "type" keyword, union types (String | Number), and optional types (String?).
func callBuiltinCoerce(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("as operator requires exactly 2 arguments: value, type", e.Pos())
        }

        value, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        var typeName string

        switch t := e.Args[1].(type) {
        case *ast.Ident:
                typeName = t.Name

        case *ast.BasicLit:
                if t.Kind != token.STRING {
                        return nil, newPosError("as operator type argument must be a string literal or identifier", t.Pos())
                }
                unquoted, err := strconv.Unquote(t.Value)
                if err != nil {
                        return nil, newPosError(fmt.Sprintf("invalid type string literal: %v", err), t.Pos())
                }
                typeName = unquoted

        default:
                v, err := evalASTWithDepth(e.Args[1], context, depth)
                if err != nil {
                        return nil, err
                }
                s, ok := v.(string)
                if !ok {
                        return nil, newPosError("as operator type argument must evaluate to a string", e.Args[1].Pos())
                }
                typeName = s
        }

        return coerceToType(value, typeName, context, e.Args[1].Pos())
}

func callBuiltinCase(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("__case requires exactly 2 arguments: value, cases", e.Pos())
        }

        val, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        casesAST, ok := e.Args[1].(*ast.CompositeLit)
        if !ok {
                return nil, newPosError("__case expects an array literal for its second argument", e.Args[1].Pos())
        }

        for _, elt := range casesAST.Elts {
                // Each element is map[string]interface{}{"pattern": "...", "result": ...}
                caseMapAST, ok := elt.(*ast.CompositeLit)
                if !ok {
                        continue
                }

                var patternStr string
                var resultAST ast.Expr

                for _, kvElt := range caseMapAST.Elts {
                        kv, ok := kvElt.(*ast.KeyValueExpr)
                        if !ok {
                                continue
                        }
                        keyIdent, ok := kv.Key.(*ast.BasicLit)
                        if !ok || keyIdent.Kind != token.STRING {
                                continue
                        }
                        key, _ := strconv.Unquote(keyIdent.Value)

                        if key == "pattern" {
                                if lit, ok := kv.Value.(*ast.BasicLit); ok && lit.Kind == token.STRING {
                                        patternStr, _ = strconv.Unquote(lit.Value)
                                }
                        } else if key == "result" {
                                resultAST = kv.Value
                        }
                }

                if patternStr == "" || resultAST == nil {
                        continue
                }

                // Check if pattern matches
                matchContext := copyContext(context)
                matched, matchFound, err := matchPattern(val, patternStr, matchContext, e.Pos())
                if err != nil {
                        return nil, err
                }

                if matchFound {
                        return evalASTWithDepth(resultAST, matched, depth+1)
                }
        }

        return nil, newPosError("no case matched the value", e.Pos())
}

func matchPattern(val interface{}, patternStr string, context map[string]interface{}, pos token.Pos) (map[string]interface{}, bool, error) {
        if patternStr == "else" {
                return context, true, nil
        }

        // Handle 's is String if s > 5' or 'is String' or literal
        p := strings.TrimSpace(patternStr)

        // Check for 'if' condition
        ifParts := strings.SplitN(p, " if ", 2)
        p = strings.TrimSpace(ifParts[0])

        matched := false
        matchContext := context

        // 1. Literal match (try parsing as basic literal)
        if !strings.Contains(p, " is ") && p != "else" {
                litAST, err := goparser.ParseExpr(p)
                if err == nil {
                        litVal, err := evalAST(litAST, nil)
                        if err == nil {
                                if litVal == val {
                                        matched = true
                                }
                        }
                }
        }

        // 2. Type match: [varName] is TypeExpr
        isIdx := strings.Index(p, " is ")
        startsIs := strings.HasPrefix(p, "is ")
        if !matched && (isIdx != -1 || startsIs) {
                var varName, typeExpr string
                if startsIs {
                        varName = ""
                        typeExpr = strings.TrimSpace(p[3:])
                } else {
                        parts := strings.SplitN(p, " is ", 2)
                        varName = strings.TrimSpace(parts[0])
                        typeExpr = strings.TrimSpace(parts[1])
                }

                if varName == "" {
                        // 'is String'
                        if matchesTypeExactly(val, typeExpr, context) || getTypeName(val) == typeExpr {
                                matched = true
                        }
                } else {
                        // 's is String'
                        if matchesTypeExactly(val, typeExpr, context) || getTypeName(val) == typeExpr {
                                matched = true
                                matchContext = copyContext(context)
                                matchContext[varName] = val
                        }
                }
        }

        if !matched {
                return nil, false, nil
        }

        // 3. Evaluate 'if' condition if present
        if len(ifParts) > 1 {
                condStr := strings.TrimSpace(ifParts[1])
                condAST, err := goparser.ParseExpr(condStr)
                if err != nil {
                        return nil, false, newPosError(fmt.Sprintf("invalid case condition: %v", err), pos)
                }
                condVal, err := evalASTWithDepth(condAST, matchContext, 0)
                if err != nil {
                        return nil, false, err
                }
                condBool, ok := condVal.(bool)
                if !ok {
                        return nil, false, newPosError(fmt.Sprintf("case condition must be boolean, got %T", condVal), pos)
                }
                if !condBool {
                        return nil, false, nil
                }
        }

        return matchContext, true, nil
}

// callMust implements the must(value, matcher(s)) function
// It can take 2 arguments (value, matcher) or (value, array of matchers)
func callMust(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) < 2 || len(e.Args) > 2 {
                return nil, newPosError("must() requires exactly 2 arguments: value and matcher(s)", e.Pos())
        }

        // Evaluate the value
        value, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        // Evaluate the matcher(s)
        matcherArg, err := evalASTWithDepth(e.Args[1], context, depth)
        if err != nil {
                return nil, err
        }

        // Handle single matcher or array of matchers
        var matchers []Matcher
        switch m := matcherArg.(type) {
        case Matcher:
                matchers = []Matcher{m}
        case []interface{}:
                // Array of matchers
                for i, item := range m {
                        matcher, ok := item.(Matcher)
                        if !ok {
                                return nil, newPosError(fmt.Sprintf("matcher at index %d is not a valid matcher", i), e.Args[1].Pos())
                        }
                        matchers = append(matchers, matcher)
                }
        default:
                return nil, newPosError(fmt.Sprintf("expected matcher or array of matchers, got %T", matcherArg), e.Args[1].Pos())
        }

        // Apply all matchers
        for _, matcher := range matchers {
                result := matcher(value)
                if !result.Success {
                        return nil, newPosError(fmt.Sprintf("assertion failed: %s", result.Message), e.Pos())
                }
        }

        // Return the MATCHED result on success
        return MATCHED, nil
}

// callBuiltinOrderBy implements the orderBy(array, lambda) function.

func callBuiltinOrderBy(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {

                return nil, newPosError("orderBy requires exactly 2 arguments: array, lambda", e.Pos())

        }

                return nil, newPosError(fmt.Sprintf("orderBy lambda must have 1 or 2 parameters, got %d", len(lambda.ParamNames)), e.Args[1].Pos())
        }

        // Create a slice of elements with their computed sort keys
        type elementWithKey struct {
                element interface{}
                key     interface{}
                index   int
        }

        elements := make([]elementWithKey, len(array))
        for i, elem := range array {
                lambdaContext := copyContext(context)
                lambdaContext[lambda.ParamNames[0]] = elem
                if len(lambda.ParamNames) > 1 {
                        lambdaContext[lambda.ParamNames[1]] = i
                }

                key, err := evalASTWithDepth(lambda.BodyAST, lambdaContext, depth+1)
                if err != nil {
                        return nil, err
                }

                elements[i] = elementWithKey{
                        element: elem,
                        key:     key,
                        index:   i,
                }
        }

        // Sort the elements by their keys
        sort.Slice(elements, func(i, j int) bool {
                cmp, err := compareValues(elements[i].key, elements[j].key)

// compileRegex compiles a regex pattern with optional flags.
func compileRegex(pattern string, flags string) (*regexp.Regexp, error) {
        prefix := ""
        if strings.Contains(flags, "i") {
                prefix += "(?i)"
        }
        if strings.Contains(flags, "m") {
                prefix += "(?m)"
        }
        if strings.Contains(flags, "s") {
                prefix += "(?s)"
        }
        return regexp.Compile(prefix + pattern)
}

// callBuiltinMatch implements the match(string, regex) function.
// Returns capture groups from the first match.
func callBuiltinMatch(args []interface{}, e *ast.CallExpr) (interface{}, error) {
        if len(args) < 2 || len(args) > 3 {
                return nil, newPosError("match requires 2 or 3 arguments: string, regex, [flags]", e.Pos())
        }

        text, ok := args[0].(string)
        if !ok {
                return nil, newPosError(fmt.Sprintf("match expects first argument to be a string, got %T", args[0]), e.Pos())
        }

        pattern, ok := args[1].(string)
        if !ok {
                return nil, newPosError(fmt.Sprintf("match expects second argument to be a string, got %T", args[1]), e.Pos())
        }

        flags := ""
        if len(args) == 3 {
                f, ok := args[2].(string)
                if !ok {
                        return nil, newPosError(fmt.Sprintf("match expects third argument (flags) to be a string, got %T", args[2]), e.Pos())
                }
                flags = f
        }

        // Compile the regex
        re, err := compileRegex(pattern, flags)
        if err != nil {
                return nil, newPosError(fmt.Sprintf("invalid regex pattern: %s", err), e.Pos())
        }

        // Find the first match
        matches := re.FindStringSubmatch(text)
        if matches == nil {
                return nil, nil
        }

        // Return the capture groups (excluding the full match at index 0)
        result := make([]interface{}, len(matches)-1)
        for i := 1; i < len(matches); i++ {
                result[i-1] = matches[i]
        }
        return result, nil
}

                        result[i] = match[0]

                        groups := make([]interface{}, len(match)-1)
                        for j := 1; j < len(match); j++ {
                                groups[j-1] = match[j]
                        }
                        result[i] = groups
                }
        }

        return result, nil
}

// callBuiltinMapObject implements the mapObject(object, lambda) function.
// Transforms object keys/values using a lambda function.
// The lambda receives (key, value) and returns [newKey, newValue].
func callBuiltinMapObject(e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(e.Args) != 2 {
                return nil, newPosError("mapObject requires exactly 2 arguments: object, lambda", e.Pos())
        }

        // Evaluate the object argument
        objVal, err := evalASTWithDepth(e.Args[0], context, depth)
        if err != nil {
                return nil, err
        }

        // Check that the first argument is an object
        obj, ok := objVal.(map[string]interface{})
        if !ok {
                return nil, newPosError(fmt.Sprintf("mapObject expects an object, got %T", objVal), e.Args[0].Pos())
        }

        // Evaluate the lambda argument to get the lambda function
        lambdaVal, err := evalASTWithDepth(e.Args[1], context, depth)
        if err != nil {
                return nil, err
        }

        lambda, ok := lambdaVal.(*Lambda)
        if !ok {
                return nil, newPosError(fmt.Sprintf("mapObject expects a lambda function, got %T", lambdaVal), e.Args[1].Pos())
        }

        // The lambda should have 2 parameters (key, value)
        if len(lambda.ParamNames) != 2 {
                return nil, newPosError(fmt.Sprintf("mapObject lambda must have exactly 2 parameters, got %d", len(lambda.ParamNames)), e.Args[1].Pos())
        }

        result := make(map[string]interface{})

        // Create a deterministic order by sorting keys
        keys := make([]string, 0, len(obj))
        for k := range obj {
                keys = append(keys, k)
        }
        sort.Strings(keys)

        for _, key := range keys {
                value := obj[key]
                lambdaContext := copyContext(context)
                lambdaContext[lambda.ParamNames[0]] = key
                lambdaContext[lambda.ParamNames[1]] = value

                mapResult, err := evalASTWithDepth(lambda.BodyAST, lambdaContext, depth+1)
                if err != nil {
                        return nil, err
                }

                // The result should be an array [newKey, newValue]
                resultArray, ok := mapResult.([]interface{})
                if !ok || len(resultArray) != 2 {
                        return nil, newPosError("mapObject lambda must return an array of [key, value]", e.Args[1].Pos())
                }

                newKey, ok := resultArray[0].(string)
                if !ok {
                        return nil, newPosError(fmt.Sprintf("mapObject lambda key must be a string, got %T", resultArray[0]), e.Args[1].Pos())
                }

                result[newKey] = resultArray[1]
        }

        return result, nil
}

        if len(args) != 0 {
                return nil, newPosError("random takes no arguments", e.Pos())
        }

        // Get a random number between 0 and 1
        randomBytes := make([]byte, 8)
        _, err := rand.Read(randomBytes)
        if err != nil {
                return nil, newPosError(fmt.Sprintf("failed to generate random number: %s", err), e.Pos())
        }

        // Convert to a float between 0 and 1
        randomInt := int64(0)

        return randomFloat, nil
}

// callBuiltinConcat implements the __concat(value1, value2, ...) function.
// Concatenates strings, arrays, and objects.
func callBuiltinConcat(args []interface{}, e *ast.CallExpr) (interface{}, error) {
        if len(args) < 2 {
                return nil, newPosError("concat requires at least 2 arguments", e.Pos())
        }

        // Check if we're concatenating strings
        allStrings := true
        for _, arg := range args {
                if _, ok := arg.(string); !ok {
                        allStrings = false
                        break
                }
        }

        if allStrings {
                var result strings.Builder
                for _, arg := range args {
                        result.WriteString(arg.(string))
                }
                return result.String(), nil
        }

        // Check if we're concatenating arrays
        allArrays := true
        for _, arg := range args {
                if _, ok := arg.([]interface{}); !ok {
                        allArrays = false
                        break
                }
        }

        if allArrays {
                var totalLen int
                for _, arg := range args {
                        totalLen += len(arg.([]interface{}))
                }
                result := make([]interface{}, 0, totalLen)
                for _, arg := range args {
                        result = append(result, arg.([]interface{})...)
                }
                return result, nil
        }

        // Check if we're concatenating objects
        allObjects := true
        for _, arg := range args {
                if _, ok := arg.(map[string]interface{}); !ok {
                        allObjects = false
                        break
                }
        }

        if allObjects {
                result := make(map[string]interface{})
                for _, arg := range args {
                        obj := arg.(map[string]interface{})
                        for k, v := range obj {
                                result[k] = v

        Mapping   []int  // Mapping from transformed indices back to original

        FullRaw    string // The full raw input for line/column calculation

// Lambda represents a lambda function with its parameter names and body.

        ParamNames []string               // Names of parameters (e.g., ["x"] or ["item", "index"])
        Body       string                 // The body expression as a string (unevaluated)
        BodyAST    ast.Expr               // The body expression as an AST (for evaluation with parameters)

// MarshalJSON implements json.Marshaler for Lambda.
func (l *Lambda) MarshalJSON() ([]byte, error) {
        // Represent lambda as a string in JSON output for display purposes
        paramList := strings.Join(l.ParamNames, ", ")
        repr := fmt.Sprintf("(lambda: [%s] => %s)", paramList, l.Body)
        return []byte(fmt.Sprintf("\"%s\"", repr)), nil
}

// Evaluate parses and evaluates the expression.
func Evaluate(exprStr string, context map[string]interface{}, mapping []int, bodyOffset int, fullRaw string) (interface{}, error) {
        ctx := &ErrorContext{
                ExprStr:    exprStr,
                Mapping:    mapping,
                BodyOffset: bodyOffset,
                FullRaw:    fullRaw,
        }
        return EvaluateWithContext(exprStr, context, ctx)
}

// EvaluateWithContext parses and evaluates the expression with error context.
func EvaluateWithContext(exprStr string, context map[string]interface{}, errCtx *ErrorContext) (interface{}, error) {
        expr, err := parser.ParseExpr(exprStr)
        if err != nil {
                return nil, errCtx.FormatParseError(err, exprStr)
        }

        result, err := evalASTWithDepth(expr, context, 0)
        if err != nil {
                return nil, errCtx.FormatEvalError(err)
        }
        return result, nil
}

// FormatEvalError converts a posError to a user-friendly error with line/column information
func (ec *ErrorContext) FormatEvalError(err error) error {
        var pe *posError
        if !errors.As(err, &pe) {
                return err
        }

        // Calculate absolute offset in exprStr (parser positions are 1-based)
        offset := int(pe.pos) - 1
        if offset < 0 {
                return err
        }
        if offset >= len(ec.Mapping) {
                offset = len(ec.Mapping) - 1
        }

        originalOffset := ec.Mapping[offset] + ec.BodyOffset
        origLine, origCol := offsetToLineCol(ec.FullRaw, originalOffset)

        return fmt.Errorf("%d:%d: %s", origLine, origCol, pe.msg)
}

// FormatParseError converts a parser error to a user-friendly error with line/column information
func (ec *ErrorContext) FormatParseError(err error, exprStr string) error {
        // Try to extract position from error message (Go parser errors: "1:12: expected ...")
        errMsg := err.Error()
        var line, col int
        if _, scanErr := fmt.Sscanf(errMsg, "%d:%d:", &line, &col); scanErr != nil {
                return err
        }

        // Calculate offset in transformed string (line 1 based)
        offset := 0
        lines := strings.Split(exprStr, "\n")
        for i := 0; i < line-1; i++ {
                offset += len(lines[i]) + 1
        }
        offset += col - 1

        if offset >= len(ec.Mapping) {
                offset = len(ec.Mapping) - 1
        }

        originalOffset := ec.Mapping[offset] + ec.BodyOffset
        origLine, origCol := offsetToLineCol(ec.FullRaw, originalOffset)

        return fmt.Errorf("%d:%d: %s", origLine, origCol, errMsg[strings.Index(errMsg, ": ")+2:])
}

// offsetToLineCol converts a byte offset in a string to 1-based line and column numbers.
func offsetToLineCol(s string, offset int) (line, col int) {
        line, col = 1, 1
        for i := 0; i < offset && i < len(s); i++ {
                if s[i] == '\n' {
                        line++
                        col = 1
                } else {
                        col++
                }
        }
        return line, col
}

func evalBasicLit(e *ast.BasicLit) (interface{}, error) {
        switch e.Kind {
        case token.STRING:
                v, err := strconv.Unquote(e.Value)
                if err != nil {
                        return nil, newPosError(fmt.Sprintf("invalid string literal: %s", err), e.Pos())
                }
                return v, nil
        case token.INT:
                v, err := strconv.Atoi(e.Value)
                if err != nil {
                        return nil, newPosError(fmt.Sprintf("invalid integer literal: %s", err), e.Pos())
                }
                return v, nil
        case token.FLOAT:
                v, err := strconv.ParseFloat(e.Value, 64)
                if err != nil {
                        return nil, newPosError(fmt.Sprintf("invalid float literal: %s", err), e.Pos())
                }
                return v, nil
        default:
                return e.Value, nil
        }
}

// evalIndex evaluates an index operation: obj[idx]
func evalIndex(obj interface{}, idx interface{}, pos token.Pos) (interface{}, error) {
        if obj == nil {
                return nil, nil
        }

        switch v := obj.(type) {
        case []interface{}:
                i, ok := idx.(int)
                if !ok {
                        return nil, newPosError(fmt.Sprintf("array index must be an integer, got %T", idx), pos)
                }
                if i < 0 || i >= len(v) {
                        return nil, newPosError(fmt.Sprintf("array index out of bounds: %d", i), pos)
                }
                return v[i], nil
        case map[string]interface{}:
                key, ok := idx.(string)
                if !ok {
                        return nil, newPosError(fmt.Sprintf("map key must be a string, got %T", idx), pos)
                }
                return v[key], nil
        default:
                return nil, newPosError(fmt.Sprintf("cannot index into %T", obj), pos)
        }
}

// evalBinaryOp evaluates a binary operation and wraps errors with position information.
func evalBinaryOp(left, right interface{}, op token.Token, pos token.Pos) (interface{}, error) {
        var res interface{}
        var err error

        // Arithmetic operations
        switch op {
        case token.ADD:
                res, err = add(left, right)
        case token.SUB:
                res, err = sub(left, right)
        case token.MUL:
                res, err = mul(left, right)
        case token.QUO:
                res, err = quo(left, right)
        case token.EQL:
                return left == right, nil
        case token.NEQ:
                return left != right, nil
        // Comparison operations
        case token.LSS:
                res, err = lss(left, right)
        case token.GTR:
                res, err = gtr(left, right)
        case token.LEQ:
                res, err = leq(left, right)
        case token.GEQ:
                res, err = geq(left, right)
        // Logical operations
        case token.LAND:
                res, err = land(left, right)
        default:
                return nil, newPosError(fmt.Sprintf("unsupported operation: %v %v %v", left, op, right), pos)
        }

        if err != nil {
                return nil, newPosError(err.Error(), pos)
        }
        return res, nil
}

// evalUnaryOp evaluates a unary operation and wraps errors with position information.
func evalUnaryOp(val interface{}, op token.Token, pos token.Pos) (interface{}, error) {
        var res interface{}
        var err error

        switch op {
        case token.SUB:
                // Unary negation
                switch v := val.(type) {
                case float64:
                        res = -v
                case int:
                        res = -v
                default:
                        err = fmt.Errorf("cannot negate %T", val)
                }
        case token.NOT:
                // Logical NOT
                if v, ok := val.(bool); ok {
                        res = !v
                } else {
                        err = fmt.Errorf("cannot apply NOT to %T", val)
                }
        default:
                return nil, newPosError(fmt.Sprintf("unsupported unary operation: %v %v", op, val), pos)
        }

        if err != nil {
                return nil, newPosError(err.Error(), pos)
        }
        return res, nil
}

func validateFloat(v float64, op string) error {
        if math.IsNaN(v) {
                return fmt.Errorf("%s resulted in NaN", op)
        }
        if math.IsInf(v, 0) {
                return fmt.Errorf("%s resulted in infinity", op)
        }
        return nil
}

// numericOp performs a numeric operation with automatic type coercion.
// intOp is used for int+int, floatOp for all other numeric combinations.
// tryIntIntOp attempts to apply an int-to-int operation and returns (result, ok, error).
func tryIntIntOp(left, right interface{}, intOp func(int, int) int) (interface{}, bool, error) {
        l, okL := left.(int)
        r, okR := right.(int)
        if okL && okR {
                return intOp(l, r), true, nil
        }
        return nil, false, nil
}

// tryFloatOp attempts to apply a float operation and returns (result, ok, error).
// It handles all float + int, int + float, and float + float combinations.
func tryFloatOp(left, right interface{}, floatOp func(float64, float64) float64, opName string) (interface{}, bool, error) {
        var l, r float64
        var okL, okR bool

        if il, ok := left.(int); ok {
                l, okL = float64(il), true
        } else if fl, ok := left.(float64); ok {
                l, okL = fl, true
        }

        if ir, ok := right.(int); ok {
                r, okR = float64(ir), true
        } else if fr, ok := right.(float64); ok {
                r, okR = fr, true
        }

        if okL && okR {
                result := floatOp(l, r)
                if err := validateFloat(result, opName); err != nil {
                        return nil, false, err
                }
                return result, true, nil
        }

        return nil, false, nil
}

func numericOp(left, right interface{}, intOp func(int, int) int, floatOp func(float64, float64) float64, opName, verb string) (interface{}, error) {
        // Try int + int
        result, ok, err := tryIntIntOp(left, right, intOp)
        if ok || err != nil {
                return result, err
        }

        // Try mixed int/float operations
        result, ok, err = tryFloatOp(left, right, floatOp, opName)
        if ok || err != nil {
                return result, err
        }

        return nil, fmt.Errorf("cannot %s %T and %T", verb, left, right)
}

func add(left, right interface{}) (interface{}, error) {
        // Special case: string concatenation (supports string + any and any + string)
        if l, okL := left.(string); okL {
                if r, okR := right.(string); okR {
                        return l + r, nil
                }
                // String + non-string: convert right to string
                return l + fmt.Sprintf("%v", right), nil
        }
        if r, okR := right.(string); okR {
                // Non-string + string: convert left to string
                return fmt.Sprintf("%v", left) + r, nil
        }
        
        // Special case: array concatenation (supports array + array)
        if l, okL := left.([]interface{}); okL {
                if r, okR := right.([]interface{}); okR {
                        result := make([]interface{}, 0, len(l)+len(r))
                        result = append(result, l...)
                        result = append(result, r...)
                        return result, nil
                }
        }
        
        return numericOp(left, right,
                func(l, r int) int { return l + r },
                func(l, r float64) float64 { return l + r },
                "addition", "add")
}

func sub(left, right interface{}) (interface{}, error) {
        return numericOp(left, right,
                func(l, r int) int { return l - r },
                func(l, r float64) float64 { return l - r },
                "subtraction", "subtract")
}

func mul(left, right interface{}) (interface{}, error) {
        return numericOp(left, right,
                func(l, r int) int { return l * r },
                func(l, r float64) float64 { return l * r },
                "multiplication", "multiply")
}

func quo(left, right interface{}) (interface{}, error) {
        // Check for division by zero before calling numericOp
        if r, ok := right.(int); ok && r == 0 {
                return nil, fmt.Errorf("division by zero")
        }
        if r, ok := right.(float64); ok && r == 0 {
                return nil, fmt.Errorf("division by zero")
        }
        return numericOp(left, right,
                func(l, r int) int { return l / r },
                func(l, r float64) float64 { return l / r },
                "division", "divide")
}

func land(left, right interface{}) (interface{}, error) {
        l, okL := left.(bool)
        r, okR := right.(bool)
        if okL && okR {
                return l && r, nil
        }
        return nil, fmt.Errorf("logical AND requires booleans, got %T and %T", left, right)
}

// numericCompare performs a numeric comparison with automatic type coercion.
// op function should return the result of comparing two float64 values.
func numericCompare(left, right interface{}, op func(float64, float64) bool, opName string) (interface{}, error) {
        // Try int-int comparison first
        if l, okL := left.(int); okL {
                if r, okR := right.(int); okR {
                        return op(float64(l), float64(r)), nil
                }
                if r, okR := right.(float64); okR {
                        return op(float64(l), r), nil
                }
        }
        // Try float comparisons
        if l, okL := left.(float64); okL {
                if r, okR := right.(float64); okR {
                        return op(l, r), nil
                }
                if r, okR := right.(int); okR {
                        return op(l, float64(r)), nil
                }
        }
        return nil, fmt.Errorf("cannot compare %T and %T with %s", left, right, opName)
}

func lss(left, right interface{}) (interface{}, error) {
        return numericCompare(left, right, func(l, r float64) bool { return l < r }, "<")
}

func gtr(left, right interface{}) (interface{}, error) {
        return numericCompare(left, right, func(l, r float64) bool { return l > r }, ">")
}

func leq(left, right interface{}) (interface{}, error) {
        return numericCompare(left, right, func(l, r float64) bool { return l <= r }, "<=")
}

func geq(left, right interface{}) (interface{}, error) {
        return numericCompare(left, right, func(l, r float64) bool { return l >= r }, ">=")
}

// TypeDef represents a custom type definition declared with the 'type' keyword.
// Example: type Currency = String { format: "##.00" }
type TypeDef struct {
        Name       string                 // The type name (e.g., "Currency")
        BaseType   string                 // The base type (e.g., "String", "Number")
        Properties map[string]interface{} // Optional properties (e.g., format, locale)
}

// MarshalJSON implements json.Marshaler for TypeDef.
func (t *TypeDef) MarshalJSON() ([]byte, error) {
        repr := fmt.Sprintf("(type: %s = %s)", t.Name, t.BaseType)
        return []byte(fmt.Sprintf("\"%s\"", repr)), nil
}

// getTypeName returns a DataWeave-compatible type name for a value.
func getTypeName(v interface{}) string {
        switch v.(type) {
        case nil:
                return "Null"
        case bool:
                return "Boolean"
        case int:
                return "Number"
        case float64:
                return "Number"
        case string:
                return "String"
        case []interface{}:
                return "Array"
        case map[string]interface{}:
                return "Object"
        case *Lambda:
                return "Function"
        case *TypeDef:
                return "Type"
        default:
                return "Unknown"
        }
}

// parseOptional strips trailing '?' and returns (base, isOptional).
// Returns an error message if the optional syntax is invalid (e.g., "?").
func parseOptional(typeExpr string) (base string, isOptional bool, errMsg string) {
        trimmed := strings.TrimSpace(typeExpr)
        if !strings.HasSuffix(trimmed, "?") {
                return trimmed, false, ""
        }
        base = strings.TrimSpace(trimmed[:len(trimmed)-1])
        if base == "" {
                return "", true, fmt.Sprintf("invalid optional type: %q", typeExpr)
        }
        return base, true, ""
}

// splitUnion splits "T1 | T2 | T3" into cleaned, trimmed type names.
func splitUnion(typeExpr string) []string {
        parts := strings.Split(typeExpr, "|")
        out := make([]string, 0, len(parts))
        for _, p := range parts {
                if s := strings.TrimSpace(p); s != "" {
                        out = append(out, s)
                }
        }
        return out
}

// cloneVisited creates a shallow copy of the visited map for cycle detection.
func cloneVisited(in map[string]bool) map[string]bool {
        if in == nil {
                return nil
        }
        out := make(map[string]bool, len(in))
        for k, v := range in {
                out[k] = v
        }
        return out
}

// coerceToType converts a value to the specified type.
func coerceToType(value interface{}, typeName string, context map[string]interface{}, pos token.Pos) (interface{}, error) {
        return coerceToTypeWithVisited(value, typeName, context, pos, make(map[string]bool), nil)
}

// coerceToTypeWithVisited handles type coercion with cycle detection.
// Properties are passed down from custom types for formatting/locale support.
// Supports union types (String | Number) and optional types (String?).
// Note: all members of a union share the same property map. Custom types in unions
// merge their TypeDef.Properties with the shared properties (custom properties take precedence).
func coerceToTypeWithVisited(value interface{}, typeName string, context map[string]interface{}, pos token.Pos, visited map[string]bool, properties map[string]interface{}) (interface{}, error) {
        trimmed, isOptional, errMsg := parseOptional(typeName)
        if errMsg != "" {
                return nil, newPosError(errMsg, pos)
        }
        if isOptional {
                return coerceToUnion(value, []string{trimmed, "Null"}, context, pos, visited, properties)
        }

        if strings.Contains(trimmed, "|") {
                return coerceToUnion(value, splitUnion(trimmed), context, pos, visited, properties)
        }

        switch trimmed {
        case "String":
                return coerceToStringWithProps(value, properties), nil
        case "Number":
                return coerceToNumber(value, pos)
        case "Boolean":
                return coerceToBoolean(value, pos)
        case "Null":
                if value == nil {
                        return nil, nil
                }
                return nil, newPosError(fmt.Sprintf("cannot coerce %T to Null", value), pos)
        }

        if visited[trimmed] {
                return nil, newPosError(fmt.Sprintf("cyclic type definition: %s", trimmed), pos)
        }
        visited[trimmed] = true

        if typeDef, ok := context[trimmed]; ok {
                if td, ok := typeDef.(*TypeDef); ok {
                        mergedProps := mergeProperties(td.Properties, properties)
                        return coerceToTypeWithVisited(value, td.BaseType, context, pos, visited, mergedProps)
                }
        }

        return nil, newPosError(fmt.Sprintf("unknown type: %s", trimmed), pos)
}

// coerceToUnion attempts to coerce a value to one of multiple types in a union.
// It first checks for exact type matches (no coercion needed), then tries coercion.
// Note: all union members share the same property map. For custom types inside the union,
// their own TypeDef.Properties are merged with these shared properties.
func coerceToUnion(value interface{}, typeNames []string, context map[string]interface{}, pos token.Pos, visited map[string]bool, properties map[string]interface{}) (interface{}, error) {
        cleaned := splitUnion(strings.Join(typeNames, "|"))
        if len(cleaned) == 0 {
                return nil, newPosError("empty union type", pos)
        }

        for _, name := range cleaned {
                if matchesTypeExactly(value, name, context) {
                        return value, nil
                }
        }

        var firstErr error
        for _, name := range cleaned {
                v, err := coerceToTypeWithVisited(value, name, context, pos, cloneVisited(visited), properties)
                if err == nil {
                        return v, nil
                }
                if firstErr == nil {
                        firstErr = err
                }
        }

        msg := fmt.Sprintf("cannot coerce to any of union types: %s", strings.Join(cleaned, " | "))
        if firstErr != nil {
                if pe, ok := firstErr.(*posError); ok {
                        msg += fmt.Sprintf(" (%s)", pe.msg)
                }
        }
        return nil, newPosError(msg, pos)
}

// matchesTypeExactly returns true if the value already matches the given type without coercion.
func matchesTypeExactly(value interface{}, typeName string, context map[string]interface{}) bool {
        base, isOptional, errMsg := parseOptional(typeName)
        if errMsg != "" {
                // Invalid optional syntax - doesn't match
                return false
        }

        // Handle optional (T?) - value matches if it matches T or is nil
        if isOptional {
                if value == nil {
                        return true
                }
                return matchesTypeExactly(value, base, context)
        }

        // Handle union (T1 | T2 | ...)
        if strings.Contains(base, "|") {
                parts := splitUnion(base)
                for _, part := range parts {
                        if matchesTypeExactly(value, part, context) {
                                return true
                        }
                }
                return false
        }
        trimmed := base

        switch trimmed {
        case "String":
                _, ok := value.(string)
                return ok
        case "Number":
                switch value.(type) {
                case int, float64:
                        return true
                }
                return false
        case "Boolean":
                _, ok := value.(bool)
                return ok
        case "Null":
                return value == nil
        case "Array":
                _, ok := value.([]interface{})
                return ok
        case "Object":
                _, ok := value.(map[string]interface{})
                return ok
        case "Function":
                _, ok := value.(*Lambda)
                return ok
        }

        // Check custom types
        if typeDef, ok := context[trimmed]; ok {
                if td, ok := typeDef.(*TypeDef); ok {
                        return matchesTypeExactly(value, td.BaseType, context)
                }
        }

        return false
}

// mergeProperties merges type properties, with child properties overriding parent.
func mergeProperties(parent, child map[string]interface{}) map[string]interface{} {
        switch {
        case parent == nil && child == nil:
                return nil
        case parent == nil:
                out := make(map[string]interface{}, len(child))
                for k, v := range child {
                        out[k] = v
                }
                return out
        case child == nil:
                out := make(map[string]interface{}, len(parent))
                for k, v := range parent {
                        out[k] = v
                }
                return out
        default:
                out := make(map[string]interface{}, len(parent)+len(child))
                for k, v := range parent {
                        out[k] = v
                }
                for k, v := range child {
                        out[k] = v
                }
                return out
        }
}

// coerceToString converts any value to a string.
func coerceToString(value interface{}) string {
        if value == nil {
                return "null"
        }
        return fmt.Sprintf("%v", value)
}

// coerceToStringWithProps converts a value to a string, applying format properties.
func coerceToStringWithProps(value interface{}, properties map[string]interface{}) string {
        if properties == nil {
                return coerceToString(value)
        }

        if format, ok := properties["format"].(string); ok {
                return applyStringFormat(value, format)
        }

        return coerceToString(value)
}

// applyStringFormat applies a format pattern to a value.
// Supports DataWeave-style patterns like "##.00" for numbers.
func applyStringFormat(value interface{}, format string) string {
        switch v := value.(type) {
        case int:
                return formatNumber(float64(v), format)
        case float64:
                return formatNumber(v, format)
        default:
                return coerceToString(value)
        }
}

// formatNumber formats a number using a pattern like "##.00" or "###,###.00".
func formatNumber(num float64, pattern string) string {
        decimalIdx := strings.LastIndex(pattern, ".")
        var decimals int
        var useThousandsSep bool

        if decimalIdx >= 0 {
                decimals = len(pattern) - decimalIdx - 1
        }

        if strings.Contains(pattern, ",") {
                useThousandsSep = true
        }

        if decimals > 0 {
                num = roundToDecimals(num, decimals)
        }

        intPart := int64(num)
        fracPart := num - float64(intPart)
        if fracPart < 0 {
                fracPart = -fracPart
        }

        var result strings.Builder

        if num < 0 {
                result.WriteString("-")
                intPart = -intPart
        }

        intStr := strconv.FormatInt(intPart, 10)

        if useThousandsSep {
                for i, ch := range intStr {
                        if i > 0 && (len(intStr)-i)%3 == 0 {
                                result.WriteRune(',')
                        }
                        result.WriteRune(ch)
                }
        } else {
                result.WriteString(intStr)
        }

        if decimals > 0 {
                result.WriteString(".")
                fracStr := strconv.FormatFloat(fracPart, 'f', decimals, 64)
                if len(fracStr) > 2 {
                        result.WriteString(fracStr[2:])
                } else {
                        for i := 0; i < decimals; i++ {
                                result.WriteString("0")
                        }
                }
        }

        return result.String()
}

// roundToDecimals rounds a float to the specified number of decimal places.
func roundToDecimals(num float64, decimals int) float64 {
        shift := math.Pow(10, float64(decimals))
        return math.Round(num*shift) / shift
}

// coerceToNumber converts a value to a number (int or float64).
func coerceToNumber(value interface{}, pos token.Pos) (interface{}, error) {
        switch v := value.(type) {
        case int:
                return v, nil
        case float64:
                return v, nil
        case string:
                if num, ok := ParseNumericLiteral(v); ok {
                        return num, nil
                }
                return nil, newPosError(fmt.Sprintf("cannot coerce string %q to Number", v), pos)
        case bool:
                if v {
                        return 1, nil
                }
                return 0, nil
        case nil:
                return 0, nil
        default:
                return nil, newPosError(fmt.Sprintf("cannot coerce %T to Number", value), pos)
        }
}

// coerceToBoolean converts a value to a boolean.
func coerceToBoolean(value interface{}, pos token.Pos) (interface{}, error) {
        switch v := value.(type) {
        case bool:
                return v, nil
        case string:
                if b, ok := ParseBoolLiteral(v); ok {
                        return b, nil
                }
                return nil, newPosError(fmt.Sprintf("cannot coerce string %q to Boolean", v), pos)
        case int:
                return v != 0, nil
        case float64:
                return v != 0, nil
        case nil:
                return false, nil
        default:
                return nil, newPosError(fmt.Sprintf("cannot coerce %T to Boolean", value), pos)
        }
}

// ParseNumericLiteral parses a string as int or float with strict semantics.
func ParseNumericLiteral(s string) (interface{}, bool) {
        s = strings.TrimSpace(s)
        if s == "" {
                return nil, false
        }

        if !strings.ContainsAny(s, ".eE") {
                if iv, err := strconv.Atoi(s); err == nil {
                        return iv, true
                }
        }

        if fv, err := strconv.ParseFloat(s, 64); err == nil {
                return fv, true
        }

        return nil, false
}

// ParseBoolLiteral parses "true"/"false" (case-insensitive) into bool.
func ParseBoolLiteral(s string) (bool, bool) {
        switch strings.ToLower(strings.TrimSpace(s)) {
        case "true":
                return true, true
        case "false":
                return false, true
        default:
                return false, false
        }
}

// ASTVisitor defines the visitor interface for evaluating AST nodes.
// Each Visit* method corresponds to a specific AST node type.
// All Visit* methods return (result interface{}, error).
type ASTVisitor interface {
        // VisitBasicLit visits a basic literal (string, number, etc.)
        VisitBasicLit(expr *ast.BasicLit) (interface{}, error)

        // VisitIdent visits an identifier (variable reference)
        VisitIdent(expr *ast.Ident) (interface{}, error)

        // VisitBinaryExpr visits a binary expression (a + b, a && b, etc.)
        VisitBinaryExpr(expr *ast.BinaryExpr) (interface{}, error)

        // VisitUnaryExpr visits a unary expression (e.g., -x, !x)
        VisitUnaryExpr(expr *ast.UnaryExpr) (interface{}, error)

        // VisitCompositeLit visits a composite literal (map or array)
        VisitCompositeLit(expr *ast.CompositeLit) (interface{}, error)

        // VisitParenExpr visits a parenthesized expression
        VisitParenExpr(expr *ast.ParenExpr) (interface{}, error)

        // VisitIndexExpr visits an index expression (a[b])
        VisitIndexExpr(expr *ast.IndexExpr) (interface{}, error)

        // VisitCallExpr visits a function call expression
        VisitCallExpr(expr *ast.CallExpr) (interface{}, error)
}

// DefaultVisitor provides a concrete implementation of ASTVisitor.
// It uses recursion depth tracking to prevent stack overflow attacks.
type DefaultVisitor struct {
        context map[string]interface{}
        depth   int
}

// NewDefaultVisitor creates a new DefaultVisitor with the given context and depth.
func NewDefaultVisitor(context map[string]interface{}, depth int) *DefaultVisitor {
        return &DefaultVisitor{
                context: context,
                depth:   depth,
        }
}

// Visit dispatches an AST node to the appropriate Visit* method based on its type.
// This is the entry point for the visitor pattern.
func (v *DefaultVisitor) Visit(expr ast.Expr) (interface{}, error) {
        if v.depth > MaxEvalDepth {
                return nil, newPosError("expression evaluation depth limit exceeded (possible infinite recursion)", expr.Pos())
        }

        switch e := expr.(type) {
        case *ast.BasicLit:
                return v.VisitBasicLit(e)
        case *ast.Ident:
                return v.VisitIdent(e)
        case *ast.BinaryExpr:
                return v.VisitBinaryExpr(e)
        case *ast.UnaryExpr:
                return v.VisitUnaryExpr(e)
        case *ast.CompositeLit:
                return v.VisitCompositeLit(e)
        case *ast.ParenExpr:
                return v.VisitParenExpr(e)
        case *ast.IndexExpr:
                return v.VisitIndexExpr(e)
        case *ast.CallExpr:
                return v.VisitCallExpr(e)
        }
        return nil, fmt.Errorf("unsupported expression type: %T", expr)
}

// VisitBasicLit evaluates a basic literal.
func (v *DefaultVisitor) VisitBasicLit(expr *ast.BasicLit) (interface{}, error) {
        return evalBasicLit(expr)
}

// VisitIdent evaluates an identifier (variable reference).
func (v *DefaultVisitor) VisitIdent(expr *ast.Ident) (interface{}, error) {
        return evalIdent(expr, v.context)
}

// VisitBinaryExpr evaluates a binary expression by evaluating both operands
// and applying the operator.
func (v *DefaultVisitor) VisitBinaryExpr(expr *ast.BinaryExpr) (interface{}, error) {
        childVisitor := NewDefaultVisitor(v.context, v.depth+1)

        left, err := childVisitor.Visit(expr.X)
        if err != nil {
                return nil, err
        }

        right, err := childVisitor.Visit(expr.Y)
        if err != nil {
                return nil, err
        }

        return evalBinaryOp(left, right, expr.Op, expr.Pos())
}

// VisitUnaryExpr evaluates a unary expression (e.g., -x, !x).
func (v *DefaultVisitor) VisitUnaryExpr(expr *ast.UnaryExpr) (interface{}, error) {
        childVisitor := NewDefaultVisitor(v.context, v.depth+1)

        val, err := childVisitor.Visit(expr.X)
        if err != nil {
                return nil, err
        }

        return evalUnaryOp(val, expr.Op, expr.Pos())
}

// VisitCompositeLit evaluates a composite literal (map or array).
func (v *DefaultVisitor) VisitCompositeLit(expr *ast.CompositeLit) (interface{}, error) {
        childVisitor := NewDefaultVisitor(v.context, v.depth+1)

        switch expr.Type.(type) {
        case *ast.MapType:
                res := make(map[string]interface{})
                for _, elt := range expr.Elts {
                        kv := elt.(*ast.KeyValueExpr)
                        key, err := childVisitor.Visit(kv.Key)
                        if err != nil {
                                return nil, err
                        }
                        val, err := childVisitor.Visit(kv.Value)
                        if err != nil {
                                return nil, err
                        }
                        res[fmt.Sprintf("%v", key)] = val
                }
                return res, nil
        case *ast.ArrayType:
                res := make([]interface{}, 0, len(expr.Elts))
                for _, elt := range expr.Elts {
                        val, err := childVisitor.Visit(elt)
                        if err != nil {
                                return nil, err
                        }
                        res = append(res, val)
                }
                return res, nil
        }
        return nil, newPosError(fmt.Sprintf("unsupported composite type: %T", expr.Type), expr.Pos())
}

// VisitParenExpr evaluates a parenthesized expression by visiting its inner expression.
func (v *DefaultVisitor) VisitParenExpr(expr *ast.ParenExpr) (interface{}, error) {
        childVisitor := NewDefaultVisitor(v.context, v.depth+1)
        return childVisitor.Visit(expr.X)
}

// VisitIndexExpr evaluates an index expression (a[b]).
func (v *DefaultVisitor) VisitIndexExpr(expr *ast.IndexExpr) (interface{}, error) {
        childVisitor := NewDefaultVisitor(v.context, v.depth+1)

        obj, err := childVisitor.Visit(expr.X)
        if err != nil {
                return nil, err
        }
        idx, err := childVisitor.Visit(expr.Index)
        if err != nil {
                return nil, err
        }

        return evalIndex(obj, idx, expr.Pos())
}

// VisitCallExpr evaluates a function call expression.
func (v *DefaultVisitor) VisitCallExpr(expr *ast.CallExpr) (interface{}, error) {
        // Increase depth for the function call
        childVisitor := NewDefaultVisitor(v.context, v.depth+1)
        return evalCallExprWithVisitor(expr, childVisitor)
}

// evalCallExprWithVisitor evaluates a call expression using the visitor pattern.
func evalCallExprWithVisitor(e *ast.CallExpr, visitor *DefaultVisitor) (interface{}, error) {
        fun, ok := e.Fun.(*ast.Ident)
        if !ok {
                return nil, newPosError("only simple function calls are supported", e.Fun.Pos())
        }

        // Handle special functions that need unevaluated arguments
        if handler, ok := builtinSpecials[fun.Name]; ok {
                return handler(e, visitor.context, visitor.depth)
        }

        // Helper to evaluate arguments using the visitor
        evalArgs := func() ([]interface{}, error) {
                args := make([]interface{}, 0, len(e.Args))
                for _, argExpr := range e.Args {
                        arg, err := visitor.Visit(argExpr)
                        if err != nil {
                                return nil, err
                        }
                        args = append(args, arg)
                }
                return args, nil
        }

        // Dispatch to specific function handler
        if handler, ok := builtinFunctions[fun.Name]; ok {
                args, err := evalArgs()
                if err != nil {
                        return nil, err
                }
                return handler(args, e)
        }

        // Check if it's a user-defined function in the context
        if userFn, exists := visitor.context[fun.Name]; exists {
                if lambda, ok := userFn.(*Lambda); ok {
                        args, err := evalArgs()
                        if err != nil {
                                return nil, err
                        }
                        return callUserDefinedFunctionWithVisitor(lambda, args, e, visitor.context, visitor.depth)
                }
        }

        return nil, newPosError(fmt.Sprintf("undefined function: %s", fun.Name), fun.Pos())
}

// callUserDefinedFunctionWithVisitor calls a user-defined function using the visitor pattern.
func callUserDefinedFunctionWithVisitor(lambda *Lambda, args []interface{}, e *ast.CallExpr, context map[string]interface{}, depth int) (interface{}, error) {
        if len(args) != len(lambda.ParamNames) {
                return nil, newPosError(fmt.Sprintf("function expects %d arguments, got %d", len(lambda.ParamNames), len(args)), e.Pos())
        }

        fnContext := copyContext(context)
        for i, paramName := range lambda.ParamNames {
                fnContext[paramName] = args[i]
        }

        visitor := NewDefaultVisitor(fnContext, depth+1)
        return visitor.Visit(lambda.BodyAST)
}
